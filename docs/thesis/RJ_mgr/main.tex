%% -------------------- Monospace font -------------------------
%% The class applies TeX ligatures to every font, which rewrites the straight
%% quotes of reproduced source code as typographic ones and would make the
%% listings syntactically wrong. Re-declaring the monospace font with the
%% tlig feature switched off keeps code verbatim.
\setmonofont{DejaVu Sans Mono}[Scale=MatchLowercase, RawFeature=-tlig]

%% -------------------- Float placement ------------------------
%% The class sets \raggedbottom, so a float that does not fit leaves the
